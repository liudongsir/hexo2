const exportToWord = async () => {
  userMap.isSignatur = false;

  const convertedContent = convertToWordDocument(content);

const convertToWordDocument = (content) => {
  const header = `<!DOCTYPE html><html><head><meta charset='utf-8'>
  <title>Exported Document</title></head><body style="text-align: center;">`;
  const footer = `</body></html>`;
  return `${header}${content}${footer}`;
};

// 导出为Word的函数
function exportToWordssss(element) {
  // 使用html2canvas生成div的屏幕快照
  console.log(element);
  html2canvas(element).then((canvas) => {
    console.log(canvas);
    // 创建一个空白的Word文档

    const blob = new Blob([convertToWordDocument(canvas)], {
      type: "application/msword",
    });

    // 下载Word文档
    const link = document.createElement("a");
    link.href = URL.createObjectURL(blob);
    link.download = "exported.doc";
    link.click();
  });
}

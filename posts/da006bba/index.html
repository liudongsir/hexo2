  </div>

<script setup lang="ts">
import { propTypes } from '@/utils/propTypes';
import { delOss, listByIds } from '@/api/system/oss';
import service, { globalHeaders, mergeChunks, uploadChunk } from '@/utils/request';
import { calculateFileMd5Splice } from '@/utils/md5file';


const props = defineProps({
  modelValue: {
    type: [String, Object, Array],
    default: () => []
  },
  // 数量限制
  limit: propTypes.number.def(5),
  multiple: propTypes.bool.def(true),
  // 大小限制(MB)
  fileSize: propTypes.number.def(1000),
  // 文件类型, 例如['png', 'jpg', 'jpeg']
  fileType: propTypes.array.def(['docx', 'xlsx', 'ppt', 'txt', 'pdf']),
  // all
  // 是否显示提示
  isShowTip: propTypes.bool.def(true),
  drag: propTypes.bool.def(false),
  // 启用切片上传的阈值(MB)，超过此大小的文件将使用切片上传
  sliceUploadThreshold: propTypes.number.def(50)
});
const defChunkSize = 5 * 1024 * 1024;
const { proxy } = getCurrentInstance() as ComponentInternalInstance;
const emit = defineEmits(['update:modelValue', 'uplcall']);
const number = ref(0);
const uploadList = ref<any[]>([]);

const baseUrl = import.meta.env.VITE_APP_BASE_API;
const defUploadUrl = '/resource/oss/upload';
const uploadFileUrl = ref(baseUrl + defUploadUrl); // 上传文件服务器地址
const headers = ref(globalHeaders());

const fileList = ref<any[]>([]);
const showTip = computed(() => props.isShowTip && (props.fileType || props.fileSize));

const fileUploadRef = ref<ElUploadInstance>();

watch(
  () => props.modelValue,
  async (val) => {
    if (val) {
      let temp = 1;
      // 首先将值转为数组
      let list: any[] = [];
      if (Array.isArray(val)) {
        list = val;
      } else {
        const res = await listByIds(val);
        list = res.data.map((oss) => {
          return {
            name: oss.originalName,
            url: oss.url,
            ossId: oss.ossId
          };
        });
      }
      // 然后将数组转为对象数组
      fileList.value = list.map((item) => {
        item = { name: item.name, url: item.url, ossId: item.ossId };
        item.uid = item.uid || new Date().getTime() + temp++;
        return item;
      });
    } else {
      fileList.value = [];
      return [];
    }
  },
  { deep: true, immediate: true }
);

// 上传前校检格式和大小
const handleBeforeUpload = (file: any) => {
  // 校检文件类型
  if (props.fileType.length) {
    const fileName = file.name.split('.');
    const fileExt = fileName[fileName.length - 1];
    const isTypeOk = props.fileType.indexOf(fileExt) >= 0;
    if (!isTypeOk) {
      proxy?.$modal.msgError(`文件格式不正确, 请上传${props.fileType.join('/')}格式文件!`);
      return false;
    }
  }
  // 校检文件大小
  if (props.fileSize) {
    const isLt = file.size / 1024 / 1024 < props.fileSize;
    if (!isLt) {
      proxy?.$modal.msgError(`上传文件大小不能超过 ${props.fileSize} MB!`);
      return false;
    }
  }
  number.value++;
  return true;
};


async function handleSliceUpload(option) {
  proxy?.$modal.loading('正在准备上传文件，请稍候...');
  const { file, onProgress, onSuccess, onError } = option;
  // 1. 初始化：标记文件状态为uploading，进度从0开始
  onProgress({ percent: 0 }, file);
  // 2. 计算文件MD5（分片上传必备，用于后端合并）
  const fileMd5 = await calculateFileMd5Splice(file);
  if (!fileMd5) throw new Error('MD5计算失败');
  // 3. 生成分片列表
  let cur = 0;
  const fileChunkList = [];
  let filesize = file.size;
  while (cur < filesize) {
    fileChunkList.push(file.slice(cur, cur + defChunkSize));
    cur += defChunkSize;
  }
  const totalChunks = fileChunkList.length;
  let completedChunks = 0;
  // 4. 并发上传分片，同步总进度
  // 开始上传
  const abortController = new AbortController();
  const signal = abortController.signal;
  await Promise.all(fileChunkList.map((chunk, index) => {
    return (async () => {
      let resp = await uploadChunk(index, fileMd5, chunk, signal).catch((error) => {
        abortController.abort(); // 触发取消信号，终止所有剩余任务
        throw error; // 重新抛出错误，让Promise.all捕获
      });
      if (resp.code === 200) {
        completedChunks++;
        let percent = Math.floor((completedChunks / totalChunks) * 100);
        onProgress({ percent: Math.min(percent, 99) }, file); // 留1%给合并阶段
      }
    })();
  })).catch((error) => {
    proxy?.$modal.msgError('上传文件失败：' + file.name + ': ' + error.msg);
    throw error;
  });

  // 6. 所有分片上传完成，调用合并接口
  onProgress({ percent: 99 }, file); // 合并阶段显示99%

  let res = await mergeChunks(file.name, fileMd5, totalChunks);

  if (res.code !== 200) {
    throw new Error('合并文件失败' + res.msg);
  }

  // 7. 上传完成：同步到Upload组件（更新状态为success）
  onProgress({ percent: 100 }, file);
  onSuccess(res, file);
}

async function handleSingleUpload(option) {
  proxy?.$modal.loading('正在上传文件，请稍候...');
  const { file, onProgress, onSuccess, onError } = option;
  const action = defUploadUrl;
  const formData = new FormData();
  if (option.data) {
    for (const [key, value] of Object.entries(option.data)) {
      if (Array.isArray(value) && value.length)
        formData.append(key, ...value);
      else
        formData.append(key, value);
    }
  }
  formData.append(option.filename, option.file, option.file.name);
  const res = await service.post(action,
    formData, {
      headers: { repeatSubmit: false },
      onUploadProgress: (progressEvent) => {
        const progressEvt = progressEvent;
        progressEvt.percent = progressEvent.total > 0 ? progressEvent.loaded / progressEvent.total * 100 : 0;
        onProgress(progressEvt);
      }
    });
  onSuccess(res, file);
}

const customHttpRequest = async (option) => {
  const { file, onError } = option;
  const isLargeFile = file.size / 1024 / 1024 > props.sliceUploadThreshold;
  try {
    if (isLargeFile) {
      await handleSliceUpload(option);
    } else {
      await handleSingleUpload(option);
    }
  } catch (error) {
    console.log('outer error2', error);
    // 8. 上传失败：同步到Upload组件（更新状态为fail）
    onError(error, file);
  }

};

// 文件个数超出
const handleExceed = () => {
  proxy?.$modal.msgError(`上传文件数量不能超过 ${props.limit} 个!`);
};

// 上传失败
const handleUploadError = () => {
  proxy?.$modal.msgError('上传文件失败');
  proxy?.$modal.closeLoading();
};

// 上传成功回调
const handleUploadSuccess = (res: any, file: UploadFile) => {
  if (res.code === 200) {
    uploadList.value.push({
      name: res.data.fileName,
      url: res.data.url,
      ossId: res.data.ossId
    });
    uploadedSuccessfully();
  } else {
    number.value--;
    proxy?.$modal.closeLoading();
    proxy?.$modal.msgError(res.msg);
    fileUploadRef.value?.handleRemove(file);
    uploadedSuccessfully();
  }
};

// 删除文件
const handleDelete = (index: number) => {
  let ossId = fileList.value[index].ossId;
  delOss(ossId);
  fileList.value.splice(index, 1);
  emit('update:modelValue', listToString(fileList.value));
  emit('uplcall', fileList.value);
};

// 上传结束处理
const uploadedSuccessfully = () => {
  if (number.value > 0 && uploadList.value.length === number.value) {
    fileList.value = fileList.value.filter((f) => f.url !== undefined).concat(uploadList.value);
    // todo 这里如果不修改值, 会造成handleSuccess 重复触发
    uploadList.value = [];
    number.value = 0;
    emit('update:modelValue', listToString(fileList.value));
    emit('uplcall', fileList.value);
    proxy?.$modal.closeLoading();
  }
};

// 获取文件名称
const getFileName = (name: string) => {
  // 如果是url那么取最后的名字 如果不是直接返回
  if (name.lastIndexOf('/') > -1) {
    return name.slice(name.lastIndexOf('/') + 1);
  } else {
    return name;
  }
};

// 对象转成指定字符串分隔
const listToString = (list: any[], separator?: string) => {
  let strs = '';
  separator = separator || ',';
  list.forEach((item) => {
    if (item.ossId) {
      strs += item.ossId + separator;
    }
  });
  return strs != '' ? strs.substring(0, strs.length - 1) : '';
};
</script>

import { ref } from 'vue';
import { ElMessage, ElMessageBox } from 'element-plus';


// ========== 核心配置 ==========
const uploadRef = ref(null); // Upload组件引用
const fileList = ref([]); // 原生fileList，组件自动维护
const chunkSize = 5 * 1024 * 1024; // 分片大小：5MB
const concurrency = 3; // 分片并发数：避免请求过多

/**
 * 自定义HTTP请求：覆盖原生上传，实现分片上传
 * @param {Object} options Upload组件传入的参数，包含：
 *   - file: 当前上传的文件对象（原生File）
 *   - onProgress: 进度回调（用于更新file.progress）
 *   - onSuccess: 成功回调（更新file.status为success）
 *   - onError: 失败回调（更新file.status为fail）
 */

const customHttpRequest = async (options) => {
  const { file, onProgress, onSuccess, onError } = options;
  try {
    // 1. 初始化：标记文件状态为uploading，进度从0开始
    onProgress({ percent: 0 }, file);

    // 2. 计算文件MD5（分片上传必备，用于后端合并）
    const fileMd5 = await calculateFileMD5(file, onProgress);
    if (!fileMd5) throw new Error('MD5计算失败');

    // 3. 生成分片列表
    const totalChunks = Math.ceil(file.size / chunkSize);
    const chunkList = [];
    for (let i = 0; i < totalChunks; i++) {
      const start = i * chunkSize;
      const end = Math.min(start + chunkSize, file.size);
      chunkList.push({
        index: i,
        blob: file.slice(start, end),
        size: end - start,
        totalChunks,
        fileMd5,
        fileName: file.name,
      });
    }

    // 4. 并发上传分片，同步总进度
    let completedChunks = 0;
    let running = 0; // 正在运行的并发任务数
    let currentIndex = 0; // 当前待上传的分片索引

    // 递归上传分片（控制并发）
    const uploadNextChunk = async () => {
      // 终止条件：所有分片上传完成/出错
      if (currentIndex >= chunkList.length || running < 0) return;

      running++;
      const chunk = chunkList[currentIndex];
      currentIndex++;

      try {
        // 上传单个分片
        await uploadSingleChunk(chunk, (chunkLoaded) => {
          // 5. 计算总进度：(已完成字节 + 当前分片已上传字节) / 文件总字节 * 100
          const completedBytes = completedChunks * chunkSize + chunkLoaded;
          const totalProgress = (completedBytes / file.size) * 100;
          // 同步到Upload组件原生进度：调用onProgress更新file.progress
          onProgress({ percent: Math.min(totalProgress, 99) }, file); // 留1%给合并阶段
        });

        completedChunks++;
        // 分片完成后更新进度（避免最后分片进度不更新）
        const tempProgress = (completedChunks / totalChunks) * 100;
        onProgress({ percent: Math.min(tempProgress, 99) }, file);

        // 继续上传下一个分片
        await uploadNextChunk();
      } catch (chunkError) {
        running = -1; // 标记失败，终止后续上传
        throw chunkError;
      } finally {
        if (running > 0) running--;
      }
    };

    // 启动并发上传（按配置的并发数）
    for (let i = 0; i < concurrency; i++) {
      await uploadNextChunk();
    }

    // 6. 所有分片上传完成，调用合并接口
    onProgress({ percent: 99 }, file); // 合并阶段显示99%
    const mergeResult = await mergeChunks(fileMd5, file.name, totalChunks);
    if (!mergeResult.success) throw new Error('分片合并失败');

    // 7. 上传完成：同步到Upload组件（更新状态为success）
    onProgress({ percent: 100 }, file);
    onSuccess(mergeResult, file);
    ElMessage.success(`${file.name} 上传成功`);
  } catch (error) {
    // 8. 上传失败：同步到Upload组件（更新状态为fail）
    onError(error, file);
    ElMessage.error(`${file.name} 上传失败：${error.message}`);
  }
};

/**
 * 计算文件MD5（流式计算，避免全量加载，同步进度到Upload组件）
 */
const calculateFileMD5 = async (file, onProgress) => {
  try {
    const md5 = await createMD5();
    const totalChunks = Math.ceil(file.size / chunkSize);
    let processedChunks = 0;

    for (let i = 0; i < totalChunks; i++) {
      const start = i * chunkSize;
      const end = Math.min(start + chunkSize, file.size);
      const chunkBuffer = await file.slice(start, end).arrayBuffer();
      md5.update(new Uint8Array(chunkBuffer));

      // MD5计算进度同步（占总进度的10%）
      processedChunks++;
      const md5Progress = (processedChunks / totalChunks) * 10;
      onProgress({ percent: md5Progress }, file);
    }

    const md5Str = md5.digest('hex');
    md5.free();
    return md5Str;
  } catch (error) {
    ElMessage.error('MD5计算失败：' + error.message);
    return null;
  }
};

/**
 * 上传单个分片（带进度回调）
 */
const uploadSingleChunk = async (chunk, onChunkProgress) => {
  const formData = new FormData();
  formData.append('chunk', chunk.blob);
  formData.append('chunkIndex', chunk.index);
  formData.append('fileMd5', chunk.fileMd5);
  formData.append('fileName', chunk.fileName);
  formData.append('totalChunks', chunk.totalChunks);

  return new Promise((resolve, reject) => {
    const xhr = new XMLHttpRequest();
    xhr.open('POST', '/api/upload/chunk'); // 后端分片上传接口

    // 监听单个分片的上传进度
    xhr.upload.onprogress = (e) => {
      if (e.lengthComputable) {
        // 回调当前分片已上传字节数
        onChunkProgress(e.loaded);
      }
    };

    // 分片上传完成
    xhr.onload = () => {
      if (xhr.status >= 200 && xhr.status < 300) {
        resolve(JSON.parse(xhr.responseText));
      } else {
        reject(new Error(`分片${chunk.index}上传失败：${xhr.statusText}`));
      }
    };

    // 网络错误
    xhr.onerror = () => {
      reject(new Error(`分片${chunk.index}网络请求失败`));
    };

    xhr.send(formData);
  });
};

/**
 * 调用后端合并分片接口
 */
const mergeChunks = async (fileMd5, fileName, totalChunks) => {
  try {
    const response = await fetch('/api/upload/merge', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
      },
      body: JSON.stringify({
        fileMd5,
        fileName,
        totalChunks,
      }),
    });

    const result = await response.json();
    if (!response.ok || !result.success) {
      throw new Error(result.message || '合并分片失败');
    }
    return result;
  } catch (error) {
    throw new Error('合并分片失败：' + error.message);
  }
};

        headers: {

    fileList.value = files;
    ElMessage.success('文件已删除');
  }).catch(() => {
    // 取消删除：恢复fileList
    fileList.value = [file, ...files];
  });

import CryptoJS from 'crypto-js';

const defChunkSize = 1024 * 1024; // 1MB

export const calculateFileMd5 = (file: File,chunkSize?): Promise<string> => {
  chunkSize = chunkSize || defChunkSize;
  return new Promise((resolve) => {
    let offset = 0;
    const md5Calc = CryptoJS.algo.MD5.create();
    const fileReader = new FileReader();
    fileReader.onload = (e) => {
      const wordArray = CryptoJS.lib.WordArray.create(e.target.result as ArrayBuffer);
      md5Calc.update(wordArray);
      offset += chunkSize;
      if (offset < file.size) {
        readNextChunk();
      } else {
        const md5Hash = md5Calc.finalize();
        const md5 = md5Hash.toString(CryptoJS.enc.Hex);
        resolve(md5);
      }
    };
    const readNextChunk = () => {
      const chunk = file.slice(offset, offset + chunkSize);
      fileReader.readAsArrayBuffer(chunk);
    };
    readNextChunk();
  });
};


export const calculateFileMd5Splice = (file: File,chunkSize?): Promise<string> => {
  chunkSize = chunkSize || defChunkSize;
  return new Promise((resolve,reject) => {
    const fileReader = new FileReader();
    fileReader.onload = (e) => {
      const wordArray = CryptoJS.lib.WordArray.create(e.target.result as ArrayBuffer);
      const md5Hash = CryptoJS.MD5(wordArray);
      let md5 = md5Hash.toString(CryptoJS.enc.Hex);
      resolve(md5);
    };
    // fileReader.onerror = (error) => {
    //   reject(new Error('文件读取失败：' + error?.message));
    // };
    fileReader.readAsArrayBuffer(file.slice(0, chunkSize));
  });
};

后记

全量读取文件计算md5值可能会造成oom,无论前端和后端, 有两种方法可以避免

  1. 分段读取 前后端都支持前后端分段读取,查看前面例子;
  2. 只计算部分文件的md5值, 这种会造成不同文件的md5值冲突,不利于后面的断点续传(断点续传就是根据文件的特征值查找以保存文件);
